```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import GradientBoostingRegressor
from sklearn.metrics import mean_squared_error
from sklearn.preprocessing import StandardScaler
from joblib import dump
import os

def train_and_save_model():
    # Define file path
    file_path = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-44031-seed20260915-r1\train.csv"

    # Load the data
    try:
        data = pd.read_csv(file_path)
    except FileNotFoundError:
        raise FileNotFoundError(f"The file at {file_path} was not found.")
    except Exception as e:
        raise Exception(f"An error occurred while loading the file: {e}")

    # Separate features and target
    X = data.drop(columns=['price'])
    y = data['price']

    # Split data into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(
        X, y, test_size=0.2, random_state=20260915
    )

    # Initialize and fit a preprocessing pipeline (scaling)
    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_val_scaled = scaler.transform(X_val)

    # Initialize and train a Gradient Boosting Regressor
    model = GradientBoostingRegressor(
        n_estimators=100,
        learning_rate=0.1,
        max_depth=3,
        random_state=20260915
    )

    model.fit(X_train_scaled, y_train)

    # Evaluate on validation set
    val_predictions = model.predict(X_val_scaled)
    val_rmse = np.sqrt(mean_squared_error(y_val, val_predictions))
    print(f"Validation RMSE: {val_rmse:.4f}")

    # Save the model and scaler to disk
    model_filename = "house_price_model.joblib"
    scaler_filename = "house_price_scaler.joblib"

    dump(model, model_filename)
    dump(scaler, scaler_filename)

    print(f"Model and scaler saved to {os.getcwd()} as {model_filename} and {scaler_filename}")

if __name__ == "__main__":
    train_and_save_model()
```
