```python
import os
import random

import joblib
import numpy as np
import pandas as pd

from sklearn.ensemble import RandomForestClassifier
from sklearn.impute import SimpleImputer
from sklearn.metrics import accuracy_score
from sklearn.model_selection import StratifiedKFold, train_test_split
from sklearn.pipeline import Pipeline


def main():
    # Reproducibility
    SEED = 42
    random.seed(SEED)
    np.random.seed(SEED)

    train_path = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-1464-seed20260915-r3\train.csv"
    df = pd.read_csv(train_path)

    target_col = "Class"
    feature_cols = [c for c in df.columns if c != target_col]

    X = df[feature_cols].copy()
    y = df[target_col].copy()

    # Ensure numeric inputs
    for col in feature_cols:
        X[col] = pd.to_numeric(X[col], errors="coerce")
    y = pd.to_numeric(y, errors="coerce").astype(int)

    # Hold-out validation split
    X_train, X_valid, y_train, y_valid = train_test_split(
        X,
        y,
        test_size=0.2,
        random_state=SEED,
        stratify=y,
    )

    model = Pipeline(
        steps=[
            ("imputer", SimpleImputer(strategy="median")),
            (
                "classifier",
                RandomForestClassifier(
                    n_estimators=500,
                    random_state=SEED,
                    n_jobs=-1,
                    class_weight="balanced",
                    min_samples_leaf=1,
                    min_samples_split=2,
                ),
            ),
        ]
    )

    # Validation score
    model.fit(X_train, y_train)
    y_pred = model.predict(X_valid)
    val_acc = accuracy_score(y_valid, y_pred)
    print(f"Hold-out accuracy: {val_acc:.4f}")

    # Cross-validation estimate on full data
    cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=SEED)
    cv_scores = []
    for train_idx, test_idx in cv.split(X, y):
        X_tr, X_te = X.iloc[train_idx], X.iloc[test_idx]
        y_tr, y_te = y.iloc[train_idx], y.iloc[test_idx]
        model_cv = Pipeline(
            steps=[
                ("imputer", SimpleImputer(strategy="median")),
                (
                    "classifier",
                    RandomForestClassifier(
                        n_estimators=500,
                        random_state=SEED,
                        n_jobs=-1,
                        class_weight="balanced",
                        min_samples_leaf=1,
                        min_samples_split=2,
                    ),
                ),
            ]
        )
        model_cv.fit(X_tr, y_tr)
        cv_scores.append(accuracy_score(y_te, model_cv.predict(X_te)))
    cv_scores = np.array(cv_scores)
    print(f"CV accuracy: {cv_scores.mean():.4f} ± {cv_scores.std():.4f}")

    # Refit final model on all data
    model.fit(X, y)

    artifact = {
        "model": model,
        "feature_cols": feature_cols,
        "target_col": target_col,
        "seed": SEED,
    }
    out_path = os.path.join(os.getcwd(), "trained_model.joblib")
    joblib.dump(artifact, out_path)
    print(f"Saved model to: {out_path}")


if __name__ == "__main__":
    main()
```
