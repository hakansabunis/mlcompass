```python
import os
import json
import random
import joblib
import numpy as np
import pandas as pd

from sklearn.model_selection import KFold, train_test_split
from sklearn.metrics import mean_squared_error, make_scorer
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.impute import SimpleImputer
from sklearn.ensemble import HistGradientBoostingRegressor


# Paths
TRAIN_PATH = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-44031-seed0-r1\train.csv"
MODEL_PATH = "trained_model.joblib"
METADATA_PATH = "trained_model_metadata.json"

# Reproducibility
RANDOM_STATE = 42
np.random.seed(RANDOM_STATE)
random.seed(RANDOM_STATE)


def main():
    # Load data
    df = pd.read_csv(TRAIN_PATH)

    target_col = "price"
    feature_cols = [c for c in df.columns if c != target_col]

    # Remove duplicate rows before splitting
    df = df.drop_duplicates().reset_index(drop=True)

    X = df[feature_cols].copy()
    y = df[target_col].copy()

    # Hold-out split for unbiased evaluation
    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=0.2,
        random_state=RANDOM_STATE,
        shuffle=True,
    )

    # Preprocessing
    numeric_features = feature_cols
    preprocessor = ColumnTransformer(
        transformers=[
            ("num", SimpleImputer(strategy="median"), numeric_features),
        ],
        remainder="drop",
        verbose_feature_names_out=False,
    )

    # Model
    model = HistGradientBoostingRegressor(
        loss="squared_error",
        learning_rate=0.05,
        max_iter=600,
        max_depth=None,
        max_leaf_nodes=31,
        min_samples_leaf=20,
        l2_regularization=0.0,
        max_bins=255,
        random_state=RANDOM_STATE,
    )

    pipeline = Pipeline(
        steps=[
            ("preprocess", preprocessor),
            ("model", model),
        ]
    )

    # Cross-validation on training split only
    cv = KFold(n_splits=5, shuffle=True, random_state=RANDOM_STATE)
    rmse_scorer = make_scorer(mean_squared_error, greater_is_better=False, squared=False)
    cv_scores = []
    for train_idx, val_idx in cv.split(X_train):
        X_tr, X_val = X_train.iloc[train_idx], X_train.iloc[val_idx]
        y_tr, y_val = y_train.iloc[train_idx], y_train.iloc[val_idx]
        pipeline.fit(X_tr, y_tr)
        preds = pipeline.predict(X_val)
        rmse = mean_squared_error(y_val, preds, squared=False)
        cv_scores.append(rmse)
    cv_rmse = float(np.mean(cv_scores))
    cv_rmse_std = float(np.std(cv_scores))

    # Fit on training split and evaluate on held-out test split
    pipeline.fit(X_train, y_train)
    test_preds = pipeline.predict(X_test)
    test_rmse = float(mean_squared_error(y_test, test_preds, squared=False))

    # Fit final model on all data for deployment
    final_pipeline = Pipeline(
        steps=[
            ("preprocess", preprocessor),
            ("model", model),
        ]
    )
    final_pipeline.fit(X, y)

    # Save model and metadata
    payload = {
        "model": final_pipeline,
        "feature_cols": feature_cols,
        "target_col": target_col,
        "cv_rmse": cv_rmse,
        "cv_rmse_std": cv_rmse_std,
        "test_rmse": test_rmse,
        "random_state": RANDOM_STATE,
    }
    joblib.dump(payload, MODEL_PATH)

    meta = {
        "feature_cols": feature_cols,
        "target_col": target_col,
        "cv_rmse": cv_rmse,
        "cv_rmse_std": cv_rmse_std,
        "test_rmse": test_rmse,
        "random_state": RANDOM_STATE,
        "model_path": MODEL_PATH,
    }
    with open(METADATA_PATH, "w", encoding="utf-8") as f:
        json.dump(meta, f, indent=2)

    print(f"Saved model to: {os.path.abspath(MODEL_PATH)}")
    print(f"Saved metadata to: {os.path.abspath(METADATA_PATH)}")
    print(f"5-fold CV RMSE (train split only): {cv_rmse:.6f} (+/- {cv_rmse_std:.6f})")
    print(f"Held-out test RMSE: {test_rmse:.6f}")


if __name__ == "__main__":
    main()
```
