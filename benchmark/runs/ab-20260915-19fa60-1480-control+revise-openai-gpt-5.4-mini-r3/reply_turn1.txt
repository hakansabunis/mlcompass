```python
import os
import json
import joblib
import numpy as np
import pandas as pd

from sklearn.compose import ColumnTransformer
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import OneHotEncoder, StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.model_selection import StratifiedKFold, cross_val_score
from sklearn.metrics import accuracy_score


DATA_PATH = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-1480-seed2-r3\train.csv"
MODEL_PATH = "trained_model.joblib"


def make_one_hot_encoder():
    # Compatibility across sklearn versions
    try:
        return OneHotEncoder(handle_unknown="ignore", sparse_output=False)
    except TypeError:
        return OneHotEncoder(handle_unknown="ignore", sparse=False)


def main():
    df = pd.read_csv(DATA_PATH)

    target_col = "Class"
    y = df[target_col].astype(int)
    X = df.drop(columns=[target_col])

    # Identify column types
    categorical_cols = [c for c in X.columns if X[c].dtype == "object"]
    numeric_cols = [c for c in X.columns if c not in categorical_cols]

    # Preprocessing
    numeric_preprocess = Pipeline(
        steps=[
            ("imputer", SimpleImputer(strategy="median")),
            ("scaler", StandardScaler()),
        ]
    )

    categorical_preprocess = Pipeline(
        steps=[
            ("imputer", SimpleImputer(strategy="most_frequent")),
            ("onehot", make_one_hot_encoder()),
        ]
    )

    preprocess = ColumnTransformer(
        transformers=[
            ("num", numeric_preprocess, numeric_cols),
            ("cat", categorical_preprocess, categorical_cols),
        ],
        remainder="drop",
        verbose_feature_names_out=False,
    )

    # Two candidate models; pick the better one via CV if possible.
    candidates = []

    # Good baseline for small/medium tabular data
    candidates.append(
        (
            "logreg",
            Pipeline(
                steps=[
                    ("preprocess", preprocess),
                    (
                        "model",
                        LogisticRegression(
                            max_iter=2000,
                            class_weight="balanced",
                            solver="lbfgs",
                        ),
                    ),
                ]
            ),
        )
    )

    # Often strong on tabular data and doesn't need scaling after preprocessing,
    # but works fine with the same preprocess pipeline.
    candidates.append(
        (
            "hgb",
            Pipeline(
                steps=[
                    ("preprocess", preprocess),
                    (
                        "model",
                        HistGradientBoostingClassifier(
                            learning_rate=0.05,
                            max_depth=None,
                            max_iter=300,
                            min_samples_leaf=20,
                            random_state=42,
                        ),
                    ),
                ]
            ),
        )
    )

    best_name = None
    best_score = -np.inf
    best_model = None

    # Use cross-validation for selection if there are enough rows and all classes are present.
    n_classes = y.nunique()
    cv = min(5, len(df))
    if n_classes > 1 and cv >= 2:
        # Ensure valid stratification
        class_counts = y.value_counts()
        cv = min(cv, int(class_counts.min()))
        if cv >= 2:
            skf = StratifiedKFold(n_splits=cv, shuffle=True, random_state=42)
            for name, model in candidates:
                try:
                    scores = cross_val_score(model, X, y, cv=skf, scoring="accuracy")
                    score = float(np.mean(scores))
                    if score > best_score:
                        best_score = score
                        best_name = name
                        best_model = model
                except Exception:
                    # Fall back later if a model fails
                    pass

    # If CV selection failed, just use the first model
    if best_model is None:
        best_name, best_model = candidates[0]

    # Fit final model on all data
    best_model.fit(X, y)

    # Store a little metadata alongside the fitted pipeline
    artifact = {
        "model": best_model,
        "metadata": {
            "target_column": target_col,
            "feature_columns": list(X.columns),
            "categorical_columns": categorical_cols,
            "numeric_columns": numeric_cols,
            "selected_model": best_name,
            "cv_accuracy": best_score if np.isfinite(best_score) else None,
        },
    }

    joblib.dump(artifact, MODEL_PATH)
    print(f"Saved model to {os.path.abspath(MODEL_PATH)}")


if __name__ == "__main__":
    main()
```
