```python
import os
import json
import joblib
import numpy as np
import pandas as pd

from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import StratifiedKFold, cross_val_score

# Path to the training data
DATA_PATH = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-1464-seed20260915-r2\train.csv"
MODEL_PATH = "trained_model.joblib"

def main():
    # Load data
    df = pd.read_csv(DATA_PATH)

    # Basic validation
    target_col = "Class"
    feature_cols = [c for c in df.columns if c != target_col]

    X = df[feature_cols].copy()
    y = df[target_col].copy()

    # Build a robust model pipeline
    model = Pipeline(
        steps=[
            ("imputer", SimpleImputer(strategy="most_frequent")),
            (
                "classifier",
                RandomForestClassifier(
                    n_estimators=300,
                    random_state=42,
                    n_jobs=-1,
                    class_weight=None,
                ),
            ),
        ]
    )

    # Optional quick cross-validation to ensure the model trains properly
    # (kept lightweight; if dataset is tiny, adapt number of folds)
    class_counts = y.value_counts()
    min_class_count = int(class_counts.min()) if len(class_counts) > 0 else 0
    n_splits = min(5, max(2, min_class_count)) if min_class_count >= 2 else 2

    if len(np.unique(y)) > 1 and len(df) >= n_splits:
        cv = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=42)
        _ = cross_val_score(model, X, y, cv=cv, scoring="accuracy")

    # Fit final model on all data
    model.fit(X, y)

    # Save model artifact
    artifact = {
        "model": model,
        "feature_cols": feature_cols,
        "target_col": target_col,
    }
    joblib.dump(artifact, MODEL_PATH)

    print(f"Model saved to: {os.path.abspath(MODEL_PATH)}")

if __name__ == "__main__":
    main()
```
