```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score
from joblib import dump
from tqdm import tqdm
import os

def train_and_save_model():
    # Define the path to the CSV file
    csv_path = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-1464-seed20260915-r1\train.csv"

    # Load the dataset
    print("Loading dataset...")
    df = pd.read_csv(csv_path)

    # Remove duplicates to avoid inflating held-out scores
    print("Removing duplicates...")
    df = df.drop_duplicates()

    # Split features and target
    X = df.drop(columns=['Class'])
    y = df['Class']

    # Split into train and test sets (stratified to maintain class distribution)
    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size=0.2, random_state=20260915, stratify=y
    )

    # Initialize and train the model
    print("Training model...")
    model = RandomForestClassifier(
        n_estimators=100,
        random_state=20260915,
        class_weight='balanced'  # Handle class imbalance (75% vs 25%)
    )
    model.fit(X_train, y_train)

    # Evaluate on test set
    y_pred = model.predict(X_test)
    accuracy = accuracy_score(y_test, y_pred)
    print(f"Test accuracy: {accuracy:.4f}")

    # Save the model to a file
    model_filename = "random_forest_model.joblib"
    dump(model, model_filename)
    print(f"Model saved to {os.path.abspath(model_filename)}")

if __name__ == "__main__":
    train_and_save_model()
```
