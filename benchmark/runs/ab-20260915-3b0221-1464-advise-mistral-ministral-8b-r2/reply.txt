```python
import pandas as pd
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.ensemble import RandomForestClassifier
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline
from sklearn.metrics import accuracy_score, classification_report
from joblib import dump
import os

def main():
    # Define file path
    csv_path = r"C:\Users\SABUNIS\AppData\Local\Temp\mlcab-1464-seed20260915-r2\train.csv"

    # Load the dataset
    print("Loading dataset...")
    df = pd.read_csv(csv_path)

    # Remove duplicate rows (as suggested by mlcompass)
    df = df.drop_duplicates()
    print(f"Dataset shape after removing duplicates: {df.shape}")

    # Separate features and target
    X = df.drop(columns=['Class'])
    y = df['Class']

    # Split data into training and validation sets
    X_train, X_val, y_train, y_val = train_test_split(
        X, y, test_size=0.2, random_state=20260915, stratify=y
    )

    # Create a pipeline with scaling and RandomForest
    # Using RandomForest as it handles outliers well and is robust for this binary classification task
    pipeline = Pipeline([
        ('scaler', StandardScaler()),
        ('classifier', RandomForestClassifier(
            n_estimators=100,
            random_state=20260915,
            class_weight='balanced'  # Helps with the class imbalance (75% vs 25%)
        ))
    ])

    # Train the model
    print("Training model...")
    pipeline.fit(X_train, y_train)

    # Evaluate on validation set
    y_val_pred = pipeline.predict(X_val)
    val_accuracy = accuracy_score(y_val, y_val_pred)
    print(f"\nValidation Accuracy: {val_accuracy:.4f}")
    print("\nClassification Report:")
    print(classification_report(y_val, y_val_pred))

    # Save the model to current working directory
    model_filename = "random_forest_classifier.joblib"
    dump(pipeline, model_filename)
    print(f"\nModel saved as '{model_filename}' in current directory")

if __name__ == "__main__":
    main()
```
